function filtraLead(val) {
    valMultiSelect = val.value;
    switch (val.value) {
        case "MyNuevosLeads":
            $("#leadOptions").text("Leads nuevos");
            break;
        case "todosMyLeads":
            $("#leadOptions").text("Todos los leads");
            break;
        case "myActAgendadas":
            $("#leadOptions").text("Actividades agendadas");
            break;
        case "searchEmail":
            $("#busquedabtn").text("Búsqueda por email");
            break;
        case "searchPhone":
            $("#busquedabtn").text("Búsqueda por teléfono");
            break;
        case "searchID":
            $("#busquedabtn").text("Búsqueda por ID");
            break;
        case "emptySearch":
            $("#busquedabtn").text("Búsqueda");
            $("#search-field").val("");
            break;
    }
    if (valMultiSelect === "MyNuevosLeads" || valMultiSelect === "myActVencidas" || valMultiSelect === "ActVencidas") {
        $('#FechaLead').hide();
        setFechaNueva();
    }
    else {
        $('#FechaLead').show();
        setFechaNueva();
    }
}

function ConsultaAlumnos() {
    chkEstatusAlumno();
    chkTipoAlumno();
    table = $("#GridLead").DataTable({
        destroy: true,
        //"dom": 't<"controls align-items-center"<"col info"i><"col-auto"p>>',
        "processing": true,
        "serverSide": true,
        "filter": false,
        "order": [],
        "orderMulti": false,
        "pageLength": 5,
        "lengthChange": false,
        "language": AP.gridLanguage,
        "ajax": {
            "url": AP.urlAP + "/Alumno/Consulta/Consultar",
            "type": "POST",
            "datatype": "json",
            data: function (d) {
                d.UserLeads= valMultiSelect;
                d.tipoAlumno= tipoAlumno.join(",");
                d.estatusAlumno= estatusAlumno.join(",");
                d.ProgramaEstudioId = $("#cmbProgramaEstudioId").val();
                d.ConSaldo = $("#cmbConSaldo").val();
                d.PeriodoInscripcionId = $("#cmbPeriodoInscripcionId").val();
                d.FechaHoraInicio= $("#FechaHoraInicio").val();
                d.FechaHoraFin= $("#FechaHoraFin").val();
                d.txtSearch= $("#search-field").val();
                return d;
            },
            dataSrc: function (json)
            {
                if (json.data != null)
                {
                    return json.data;
                }
                else {
                    return []
                }
                
            }
        },
        "columns": [
            { "data": "NombreCompleto", "name": "NombreCompleto", "autoWidth": true, className: "name" },
            {
                "orderable": false, "render": function (data, type, row, meta) {
                    return '<a href="#" id=' + "consulta" + row.Id + ' class="m-0" onclick="redirectConsultaLead(' + row.Id + ')" role="button" title="Ver toda la información del alumno">' + row.Email + '<br/>' + row.Telefono + '</a>';
                }
            },
            { "data": "Programa", "name": "Programa", "autoWidth": true, className: "last-activity" },
            { "data": "Matricula", "name": "Matricula", "autoWidth": true, className: "name" },
            { "data": "ConSaldo", "name": "ConSaldo", "autoWidth": true, className: "last-activity" },
            { "data": "UltimoCiclo", "name": "UltimoCiclo", "autoWidth": true, className: "last-activity" },
            { "data": "TipoAlumno", "name": "TipoAlumno", "autoWidth": true, className: "interest-level" },
            { "data": "EstatusAlumno", "name": "EstatusAlumno", "autoWidth": true, className: "interest-level" },
            //{ "data": "Periodo", "name": "Periodo", "autoWidth": true, className: "interest-level" },
            {
                "orderable": false, "render": function (data, type, row, meta) {
                    return '<a href="#" role="button" data-toggle="modal"  class="btn-icon" data-target="#lstleadActivity" onclick="ListaActividad(' + row.Id + ')" title="Ver listados de actividades" class="activities"><span class="fa fa-list"></span>';
                }
            }
        ]
    });
    showHidenColumns(table);
}

function showHidenColumns(table) {
    $('th.tools input[type="checkbox"]').on('change', function (e) {
        e.preventDefault();
        if ($(this).is(':checked')) {
            table.column($(this).attr('data-column')).visible(true);
        } else {
            table.column($(this).attr('data-column')).visible(false);
        }
    });
}

function setFechaNueva() {
    configuraFecha('#FechaHoraInicio');
    $('#FechaHoraInicio').val(DateRange(3));
    configuraFecha('#FechaHoraFin');
    $('#FechaHoraFin').val(DateRange(0));
}
function configuraFecha(tagId) {
    $(tagId).val("");
    $(tagId).datetimepicker({
        format: 'DD/MM/YYYY', locale: 'es', useCurrent: false, ignoreReadonly: true,
        icons: {
            time: "fa fa-clock-o",
            date: "fa fa-calendar-minus-o",
            up: "fa fa-arrow-up",
            down: "fa fa-arrow-down"
        }
    });
}
function getFechaLead(tagId) {
    var txtDate = $(tagId).val();
    txtDate = moment.utc(txtDate, 'DD/MM/YYYY HH:mm').format('DD/MM/YYYY HH:mm');
    $(tagId).val("");
    $(tagId).datetimepicker({
        format: 'DD/MM/YYYY HH:mm', locale: 'es', useCurrent: false, ignoreReadonly: true,
        icons: {
            time: "fa fa-clock-o",
            date: "fa fa-calendar-minus-o",
            up: "fa fa-arrow-up",
            down: "fa fa-arrow-down"
        }
    });
    $(tagId).datetimepicker('date', txtDate);
}

function ExportarCSV() {

    var UserLeads = valMultiSelect;
